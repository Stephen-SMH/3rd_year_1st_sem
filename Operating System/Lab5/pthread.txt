#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>

int sum;
void *runner(void *param);
int main(int argc, char *argv[]) {
	pthread_t tid;
	pthread_attr_t attr;
	if (argc != 2) {
		fprintf(stderr, "usage: pthread <integer value>\n");
		exit(1);
	}
	if (atoi(argv[1]) < 0) {
		fprintf(stderr, "number must be >=0\n");
		exit(2);
	}
	pthread_attr_init(&attr);
	/* create the thread */
	pthread_create(&tid, &attr, runner, argv[1]);
	int num = atoi(argv[1]);
	/* now wait for the thread to exit */
	printf("I am mother thread, I will wait for my child thread\n");
	for (int i = 1; i <= num; i++) {
	  sum += i;
	}
	pthread_join(tid, NULL);
	printf("from my child and I sum = %d\n", sum);
}

void *runner (void *param) {
	int i;
	int upper = atoi(param);
	sum = 0;
	printf("I am child thread, I am calculating\n");
	if (upper > 0) {
		for (i = 0; i <= upper * 2; ++i) {
			sum += i;
		}
	}
	pthread_exit(0);
}
